1.1 多种方式实现数组去重、扁平化、对比优缺点 ​

参考

数组去重的核心是利用数据结构（Set/Map/Object）进行唯一性判断；数组扁平化的核心是递归或迭代地将嵌套数组展开为一维数组。不同方案在时间复杂度、代码简洁度、特殊值处理上各有取舍。

数组去重的四种方案 ​

javascript
const arr = [1, 2, 2, NaN, NaN, { a: 1 }, { a: 1 }, undefined, null, null]

// 方案一：Set（最简洁，无法去重对象）
const unique1 = [...new Set(arr)] // NaN 仅保留一个，对象无法去重

// 方案二：filter + indexOf（兼容性好，NaN 会被全部剔除）
const unique2 = arr.filter((item, index) => arr.indexOf(item) === index)

// 方案三：Map（可按指定 key 对对象数组去重）
function uniqueByKey(arr, key) {
  const map = new Map()
  return arr.filter((item) => !map.has(item[key]) && map.set(item[key], true))
}

// 方案四：reduce + includes（直观，O(n²) 性能最差）
const unique4 = arr.reduce((acc, cur) => (acc.includes(cur) ? acc : [...acc, cur]), [])

数组扁平化的四种方案 ​

javascript
const nested = [1, [2, [3, [4]], 5]]

// 方案一：ES2019 Array.flat(depth)（最简洁，但参数固定）
nested.flat(Infinity) // [1, 2, 3, 4, 5]

// 方案二：递归（可控制条件，但可能栈溢出）
function flattenRecursive(arr, depth = Infinity) {
  if (depth === 0) return arr
  return arr.reduce(
    (acc, cur) => acc.concat(Array.isArray(cur) ? flattenRecursive(cur, depth - 1) : cur),
    []
  )
}

// 方案三：迭代 + 栈（避免递归栈溢出，适合深层嵌套）
function flattenStack(arr) {
  const stack = [...arr]
  const result = []
  while (stack.length) {
    const item = stack.pop()
    if (Array.isArray(item)) {
      stack.push(...item) // 展开后重新入栈
    } else {
      result.unshift(item)
    }
  }
  return result
}

// 方案四：Generator + yield*（惰性求值，适合大数组）
function* flattenGenerator(arr) {
  for (const item of arr) {
    if (Array.isArray(item)) {
      yield* flattenGenerator(item)
    } else {
      yield item
    }
  }
}

方案语法局限

方案三：手写递归深拷贝（完整版） ​

javascript
function deepClone(obj, hash = new WeakMap()) {
  // null 或非对象直接返回
  if (obj === null || typeof obj !== 'object') return obj

  // 循环引用检测
  if (hash.has(obj)) return hash.get(obj)

  // 处理 Date 和 RegExp
  if (obj instanceof Date) return new Date(obj)
  if (obj instanceof RegExp) return new RegExp(obj.source, obj.flags)

  // 处理 Map
  if (obj instanceof Map) {
    const clone = new Map()
    hash.set(obj, clone)
    obj.forEach((val, key) => clone.set(deepClone(key, hash), deepClone(val, hash)))
    return clone
  }

  // 处理 Set
  if (obj instanceof Set) {
    const clone = new Set()
    hash.set(obj, clone)
    obj.forEach((val) => clone.add(deepClone(val, hash)))
    return clone
  }

  // 处理数组和普通对象，保留原型链
  const clone = Array.isArray(obj) ? [] : Object.create(Object.getPrototypeOf(obj))
  hash.set(obj, clone)

  // 处理 Symbol key
  Reflect.ownKeys(obj).forEach((key) => {
    clone[key] = deepClone(obj[key], hash)
  })

  return clone
}

1.3 手写函数柯里化工具函数、并理解其应用场景和优势 ​

参考

柯里化（Currying） 是把一个接收多个参数的函数，变换成一系列每次只接收一个参数的函数链。核心思想是参数复用和延迟执行。

柯里化 vs 偏函数 ​

概念定义示例
柯里化将 f(a, b, c) 转为 f(a)(b)(c)，每次只接收一个参数curry(add)(1)(2)(3)
偏函数预先绑定部分参数，返回接收剩余参数的函数add.bind(null, 1) → (b,c)=>...

固定长度柯里化实现 ​

javascript
function curry(fn) {
  return function curried(...args) {
    // 参数够了就执行
    if (args.length >= fn.length) {
      return fn.apply(this, args)
    }
    // 参数不够返回新函数等待剩余参数
    return function (...rest) {
      return curried.apply(this, args.concat(rest))
    }
  }
}

// 使用
function add(a, b, c) {
  return a + b + c
}
const curriedAdd = curry(add)
curriedAdd(1)(2)(3) // 6
curriedAdd(1, 2)(3) // 6
curriedAdd(1)(2, 3) // 6

占位符支持的增强版 ​

javascript
const _ = Symbol('placeholder')

function curryWithPlaceholder(fn) {
  return function curried(...args) {
    // 剔除占位符，检查有效参数是否足够
    const validArgs = args.filter((a) => a !== _)
    if (validArgs.length >= fn.length && !args.slice(0, fn.length).includes(_)) {
      return fn.apply(this, args.slice(0, fn.length))
    }
    return function (...rest) {
      // 用新参数替换占位符
      let i = 0
      const merged = args.map((arg) => (arg === _ && i < rest.length ? rest[i++] : arg))
      return curried.apply(this, merged.concat(rest.slice(i)))
    }
  }
}

// 使用: 占位符可跳过一个位置的参数
const curried = curryWithPlaceholder(add)
curried(_, 2, 3)(1) // 6 — 先用占位符跳过a，后续再补

不定长度的"无限柯里化" ​

javascript
function infiniteCurry(fn) {
  const next = (...args) => {
    // 调用时检查是否传递了参数，没参数就计算结果
    return (...rest) => {
      if (rest.length === 0) {
        return fn(...args)
      }
      return next(...args, ...rest)
    }
  }
  return next()
}

function sum(...nums) {
  return nums.reduce((a, b) => a + b, 0)
}

1.4 手写防抖和节流工具函数、并理解其内部原理和应用场景 ​

参考

防抖（Debounce）：高频触发后，只执行最后一次，等待期间重新计时。类比电梯：有人进来就重新等。
节流（Throttle）：高频触发时，按固定频率执行。类比滴水龙头：不管来多少水，按固定间隔滴水。

防抖实现（非立即执行版 + 立即执行版） ​

javascript
// 非立即执行版：最后一次触发后等 delay 毫秒执行
function debounce(fn, delay = 300) {
  let timer = null
  return function (...args) {
    clearTimeout(timer)
    timer = setTimeout(() => {
      fn.apply(this, args)
    }, delay)
  }
}

// 立即执行版：首次触发立即执行，后续在冷却期内忽略
function debounceImmediate(fn, delay = 300) {
  let timer = null
  return function (...args) {
    const callNow = !timer
    clearTimeout(timer)
    timer = setTimeout(() => {
      timer = null
    }, delay)
    if (callNow) fn.apply(this, args)
  }
}

// 增强版：支持 immediate + cancel + 返回值（Promise版）
function debouncePro(fn, delay = 300, immediate = false) {
  let timer = null

  const debounced = function (...args) {
    return new Promise((resolve) => {
      const callNow = immediate && !timer
      clearTimeout(timer)
      timer = setTimeout(() => {
        timer = null
        if (!immediate) resolve(fn.apply(this, args))
      }, delay)
      if (callNow) resolve(fn.apply(this, args))
    })
  }

  debounced.cancel = () => {
    clearTimeout(timer)
    timer = null
  }

  return debounced
}

节流实现（时间戳版 + 定时器版） ​

javascript
// 时间戳版：第一次立即执行，最后一次不执行
function throttle(fn, interval = 300) {
  let lastTime = 0
  return function (...args) {
    const now = Date.now()
    if (now - lastTime >= interval) {
      lastTime = now
      fn.apply(this, args)
    }
  }
}

// 定时器版：第一次延迟执行，最后一次会执行
function throttleTimer(fn, interval = 300) {
  let timer = null
  return function (...args) {
    if (timer) return
    timer = setTimeout(() => {
      fn.apply(this, args)
      timer = null
    }, interval)
  }
}

// 增强版：结合两者，首次执行 + 末次也执行
function throttlePro(fn, interval = 300) {
  let lastTime = 0
  let timer = null
  return function (...args) {
    const now = Date.now()
    const remaining = interval - (now - lastTime)
    clearTimeout(timer)

    if (remaining <= 0) {
      lastTime = now
      fn.apply(this, args)
    } else {
      // 保证最后一次调用能执行
      timer = setTimeout(() => {
        lastTime = Date.now()
        fn.apply(this, args)
      }, remaining)
    }
  }
}

2.2 手动实现符合Promise/A+规范的Promise、手动实现async await ​

参考

Promise/A+ 规范的核心是状态机 + then 链式调用 + 决议过程。async/await 本质上是 Generator 函数的语法糖，配合自动执行器实现。

Promise/A+ 核心规范要点 ​

  1. 三种状态：pending → fulfilled（value），pending → rejected（reason），状态不可逆
  2. then 方法：promise.then(onFulfilled, onRejected) 必须返回一个新 Promise，实现链式调用
  3. 决议过程（Resolution Procedure）：处理 thenable 对象和循环引用
  4. 异步执行：onFulfilled/onRejected 必须异步调用（微任务）

完整 Promise 实现 ​

javascript
const PENDING = 'pending'
const FULFILLED = 'fulfilled'
const REJECTED = 'rejected'

class MyPromise {
  constructor(executor) {
    this.state = PENDING
    this.value = undefined
    this.reason = undefined
    // 存储回调数组，支持多次 then
    this.onFulfilledCallbacks = []
    this.onRejectedCallbacks = []

    const resolve = (value) => {
      // 只有 pending 状态才能转换
      if (this.state !== PENDING) return
      this.state = FULFILLED
      this.value = value
      this.onFulfilledCallbacks.forEach((fn) => fn())
    }

    const reject = (reason) => {
      if (this.state !== PENDING) return
      this.state = REJECTED
      this.reason = reason
      this.onRejectedCallbacks.forEach((fn) => fn())
    }

    try {
      executor(resolve, reject)
    } catch (err) {
      reject(err)
    }
  }

  then(onFulfilled, onRejected) {
    // 值穿透：非函数时创建默认处理器
    onFulfilled = typeof onFulfilled === 'function' ? onFulfilled : (v) => v
    onRejected =
      typeof onRejected === 'function'
        ? onRejected
        : (e) => {
            throw e
          }

    // 返回新 Promise 实现链式调用
    const promise2 = new MyPromise((resolve, reject) => {
      if (this.state === FULFILLED) {
        // 用 setTimeout 模拟微任务异步执行
        setTimeout(() => {
          try {
            const x = onFulfilled(this.value)
            resolvePromise(promise2, x, resolve, reject)
          } catch (err) {
            reject(err)
          }
        })
      }

      if (this.state === REJECTED) {
        setTimeout(() => {
          try {
            const x = onRejected(this.reason)
            resolvePromise(promise2, x, resolve, reject)
          } catch (err) {
            reject(err)
          }
        })
      }

      if (this.state === PENDING) {
        // 异步情况：将回调存入队列
        this.onFulfilledCallbacks.push(() => {
          setTimeout(() => {
            try {
              const x = onFulfilled(this.value)
              resolvePromise(promise2, x, resolve, reject)
            } catch (err) {
              reject(err)
            }
          })
        })
        this.onRejectedCallbacks.push(() => {
          setTimeout(() => {
            try {
              const x = onRejected(this.reason)
              resolvePromise(promise2, x, resolve, reject)
            } catch (err) {
              reject(err)
            }
          })
        })
      }
    })

    return promise2
  }

  catch(onRejected) {
    return this.then(null, onRejected)
  }

  finally(onFinally) {
    return this.then(
      (value) => MyPromise.resolve(onFinally()).then(() => value),
      (reason) =>
        MyPromise.resolve(onFinally()).then(() => {
          throw reason
        })
    )
  }
}

// Promise 决议过程 — 处理 thenable 和循环引用
function resolvePromise(promise2, x, resolve, reject) {
  // 防止循环引用
  if (promise2 === x) {
    return reject(new TypeError('Chaining cycle detected'))
  }

  // 防止多次调用
  let called = false

  if ((typeof x === 'object' && x !== null) || typeof x === 'function') {
    try {
      const then = x.then
      if (typeof then === 'function') {
        // x 是一个 thenable（含 then 方法的对象/函数）
        then.call(
          x,
          (y) => {
            if (called) return
            called = true
            resolvePromise(promise2, y, resolve, reject) // 递归解析
          },
          (r) => {
            if (called) return
            called = true
            reject(r)
          }
        )
      } else {
        resolve(x) // 普通对象
      }
    } catch (err) {
      if (called) return
      called = true
      reject(err)
    }
  } else {
    resolve(x) // 基本类型值
  }
}

MyPromise.reject = function (reason) {

function asyncToGenerator(generatorFn) {
  return function (...args) {
    const gen = generatorFn.apply(this, args)

    return new Promise((resolve, reject) => {
      function step(key, arg) {
        let result
        try {
          result = gen[key](arg)
        } catch (err) {
          return reject(err)
        }

        const { value, done } = result
        if (done) {
          return resolve(value)
        }
        // 将 yield 出的值包装为 Promise 继续走下一步
        return Promise.resolve(value).then(
          (val) => step('next', val),
          (err) => step('throw', err)
        )
      }

      step('next')
    })
  }
}

2.3 手写一个EventEmitter实现事件发布、订阅 ​

参考

EventEmitter（发布-订阅模式）是 Node.js 核心模块的实现，也是前端解耦通信的常用模式。核心方法：on（订阅）、off（取消）、emit（发布）、once（一次性订阅）。

完整实现 ​

javascript
class EventEmitter {
  constructor() {
    this._events = Object.create(null) // 无原型链干扰
    this._maxListeners = 10 // 默认最大监听数
  }

  // 订阅事件
  on(event, listener) {
    if (!this._events[event]) {
      this._events[event] = []
    }
    // 超出最大监听数时给出警告
    if (this._events[event].length >= this._maxListeners) {
      console.warn(`MaxListenersExceededWarning: ${event} 监听器超过 ${this._maxListeners}`)
    }
    this._events[event].push(listener)
    return this // 支持链式调用
  }

  // 一次性订阅
  once(event, listener) {
    const wrapper = (...args) => {
      this.off(event, wrapper)
      listener.apply(this, args)
    }
    wrapper._original = listener // 标记原始回调，方便 off 按原函数移除
    this.on(event, wrapper)
    return this
  }

  // 取消订阅
  off(event, listener) {
    const listeners = this._events[event]
    if (!listeners) return this

    // 不传 listener 则移除该事件全部监听器
    if (!listener) {
      delete this._events[event]
      return this
    }

    // 移除匹配的监听器（包含 once 包装的）
    this._events[event] = listeners.filter((fn) => fn !== listener && fn._original !== listener)
    return this
  }

  // 发布事件
  emit(event, ...args) {
    const listeners = this._events[event]
    if (!listeners || listeners.length === 0) {
      // error 事件的默认行为：没有监听者时抛出异常
      if (event === 'error') {
        throw args[0] instanceof Error ? args[0] : new Error(args[0])
      }
      return false
    }

    // 拷贝数组再遍历，防止在回调中修改原数组
    const listenersCopy = [...listeners]
    listenersCopy.forEach((listener) => {
      listener.apply(this, args)
    })
    return true
  }

  // 移除所有监听器
  removeAllListeners(event) {
    if (event) {
      delete this._events[event]
    } else {
      this._events = Object.create(null)
    }
    return this
  }

  // 获取事件的监听器数组
  listeners(event) {
    return this._events[event] ? [...this._events[event]] : []
  }

  // 设置最大监听数
  setMaxListeners(n) {
    this._maxListeners = n
    return this
  }
}

双向绑定 = 数据变化 → 视图更新（Model → View）+ 用户操作 → 数据更新（View → Model）。两种主流方案是 Object.defineProperty（Vue 2）和 Proxy（Vue 3）。

方案一：Object.defineProperty + 发布-订阅模式（Vue 2） ​

javascript
// 依赖收集器 — 每个响应式属性对应一个 Dep 实例
class Dep {
  constructor() {
    this.subs = [] // 订阅者（Watcher）列表
  }
  addSub(sub) {
    this.subs.push(sub)
  }
  notify() {
    this.subs.forEach((sub) => sub.update())
  }
}

// 观察者 — 数据变化时执行更新回调
class Watcher {
  constructor(vm, key, updateFn) {
    this.vm = vm
    this.key = key
    this.updateFn = updateFn
    Dep.target = this // 将当前 watcher 挂载到全局
    this.vm[this.key] // 触发 getter，收集依赖
    Dep.target = null
  }
  update() {
    this.updateFn(this.vm[this.key])
  }
}

// 将对象属性转为响应式
function observe(obj) {
  if (typeof obj !== 'object' || obj === null) return
  Object.keys(obj).forEach((key) => {
    let internalValue = obj[key]
    const dep = new Dep()

    // 递归处理嵌套对象
    observe(internalValue)

    Object.defineProperty(obj, key, {
      get() {
        if (Dep.target) dep.addSub(Dep.target) // 收集依赖
        return internalValue
      },
      set(newVal) {
        if (newVal === internalValue) return
        internalValue = newVal
        observe(newVal) // 新值也转为响应式
        dep.notify() // 通知所有订阅者更新
      }
    })
  })
}

方案二：Proxy（Vue 3） ​

function reactive(target) {
  if (typeof target !== 'object' || target === null) return target

  const depMap = new Map()

  function getDep(key) {
    let dep = depMap.get(key)
    if (!dep) {
      dep = new Dep()
      depMap.set(key, dep)
    }
    return dep
  }

  return new Proxy(target, {
    get(obj, key, receiver) {
      // 依赖收集
      if (Dep.target) getDep(key).addSub(Dep.target)
      const result = Reflect.get(obj, key, receiver)
      // 懒递归：只有访问到嵌套对象时才转为响应式
      return reactive(result)
    },

    set(obj, key, value, receiver) {
      const oldVal = obj[key]
      const result = Reflect.set(obj, key, value, receiver)
      if (oldVal !== value) {
        getDep(key).notify()
      }
      return result
    },

    deleteProperty(obj, key) {
      const had = Object.prototype.hasOwnProperty.call(obj, key)
      const result = Reflect.deleteProperty(obj, key)
      if (had) {
        getDep(key).notify()
      }
      return result
    }
  })
}

JSON.stringify 手动实现 ​

javascript
function myStringify(data) {
  const seen = new WeakSet()

  function stringify(value) {
    const type = typeof value

    // 基本类型处理
    if (type === 'string') return `"${value.replace(/"/g, '\\"')}"`
    if (type === 'number') {
      if (Number.isNaN(value) || !Number.isFinite(value)) return 'null'
      return String(value)
    }
    if (type === 'boolean') return String(value)
    if (value === null) return 'null'
    if (type === 'undefined' || type === 'function' || type === 'symbol') return undefined
    if (typeof value === 'bigint') throw new TypeError('Do not know how to serialize a BigInt')

    // Date — 转为 ISO 字符串再包裹引号
    if (value instanceof Date) return `"${value.toISOString()}"`

    // 循环引用检测
    if (typeof value === 'object') {
      if (seen.has(value)) throw new TypeError('Converting circular structure to JSON')
      seen.add(value)
    }

    // 处理 toJSON 方法（先于其他序列化逻辑）
    if (value.toJSON && typeof value.toJSON === 'function') {
      return stringify(value.toJSON())
    }

    // 数组处理
    if (Array.isArray(value)) {
      const items = value.map((item) => {
        const v = stringify(item)
        return v === undefined ? 'null' : v // 数组中 undefined 等转为 null
      })
      return `[${items.join(',')}]`
    }

    // 普通对象处理
    const pairs = []
    Object.keys(value).forEach((key) => {
      const v = stringify(value[key])
      if (v !== undefined) {
        // 值为 undefined/function/symbol 的属性直接跳过
        pairs.push(`"${key.replace(/"/g, '\\"')}":${v}`)
      }
    })
    // 注意：Symbol key 会被 JSON.stringify 自动忽略，这里用 Object.keys 已保证
    return `{${pairs.join(',')}}`
  }

  return stringify(data)
}

JSON.parse 手动实现（递归下降解析器） ​

javascript
function myParse(jsonStr) {
  const str = jsonStr.trim()
  let i = 0

  function parseValue() {
    const char = str[i]
    if (char === '"') return parseString()
    if (char === '{') return parseObject()
    if (char === '[') return parseArray()
    if (char === 't' || char === 'f') return parseBoolean()
    if (char === 'n') return parseNull()
    return parseNumber()
  }

  function parseString() {
    i++ // 跳过左引号
    let result = ''
    while (i < str.length && str[i] !== '"') {
      if (str[i] === '\\') {
        i++ // 跳过反斜杠
        const escapeMap = {
          '"': '"',
          '\\': '\\',
          '/': '/',
          b: '\b',
          f: '\f',
          n: '\n',
          r: '\r',
          t: '\t'
        }
        if (str[i] === 'u') {
          const hex = str.substring(i + 1, i + 5)
          result += String.fromCharCode(parseInt(hex, 16))
          i += 5
        } else {
          result += escapeMap[str[i]] || str[i]
          i++
        }
      } else {
        result += str[i]
        i++
      }
    }
    i++ // 跳过右引号
    return result
  }

  function parseNumber() {
    const match = str.slice(i).match(/^-?\d+(?:\.\d+)?(?:[eE][+-]?\d+)?/)
    if (!match) throw new SyntaxError('Unexpected token')
    i += match[0].length
    return Number(match[0])
  }

  function parseBoolean() {
    if (str.startsWith('true', i)) {
      i += 4
      return true
    }
    if (str.startsWith('false', i)) {
      i += 5
      return false
    }
    throw new SyntaxError('Unexpected token')
  }

  function parseNull() {
    if (str.startsWith('null', i)) {
      i += 4
      return null
    }
    throw new SyntaxError('Unexpected token')
  }

  function parseArray() {
    i++ // 跳过 [
    const result = []
    skipWhitespace()
    if (str[i] !== ']') {
      while (true) {
        result.push(parseValue())
        skipWhitespace()
        if (str[i] === ',') {
          i++
          skipWhitespace()
          continue
        }
        break
      }
    }
    i++ // 跳过 ]
    return result
  }

  function parseObject() {
    i++ // 跳过 {
    const result = {}
    skipWhitespace()
    if (str[i] !== '}') {
      while (true) {
        const key = parseString()
        skipWhitespace()
        i++ // 跳过冒号 :
        skipWhitespace()
        result[key] = parseValue()
        skipWhitespace()
        if (str[i] === ',') {
          i++
          skipWhitespace()
          continue
        }
        break
      }
    }
    i++ // 跳过 }
    return result
  }

  function skipWhitespace() {
    while (/\s/.test(str[i])) i++
  }

  skipWhitespace()
  const result = parseValue()
  skipWhitespace()
  if (i < str.length) throw new SyntaxError('Unexpected token at end')
  return result
}

2.6 手写一个模版引擎，并能解释其中原理 ​

参考

模版引擎的核心原理：模版字符串 → 解析为 Token/AST → 拼接成可执行代码 → 传入数据执行得到渲染结果。方案从简单到复杂可以分为正则替换和编译执行。

方案一：正则替换（简单变量插值） ​

javascript
function simpleTemplate(template, data) {
  // 替换 {{ }} 变量插值
  return template.replace(/\{\{\s*([\w.]+)\s*\}\}/g, (match, key) => {
    // 支持点路径访问：{{ user.name }}
    const keys = key.trim().split('.')
    let value = data
    for (const k of keys) {
      value = value?.[k]
      if (value === undefined) return ''
    }
    return value
  })
}

// 使用
const tpl = '<h1>{{ title }}</h1><p>{{ author.name }}</p>'

方案二：new Function 编译（支持逻辑控制） ​

将模版转为 JS 函数体字符串，通过 new Function 动态编译。支持 {{ }} 变量输出、{{#if}} 条件、{{#each}} 循环。

javascript
function compile(template) {
  // 核心思想：用单引号拼接字符串，遇到 {{ }} 时切换到 JS 表达式
  let code = 'let result = "";\n'

  // 将模版转为代码片段数组
  const tokens = []
  let lastIndex = 0
  // 匹配 {{ ... }} 中的内容
  const regex = /\{\{(.+?)\}\}/g
  let match

  while ((match = regex.exec(template)) !== null) {
    // 前面的纯文本部分
    if (match.index > lastIndex) {
      const text = template.slice(lastIndex, match.index)
      tokens.push({ type: 'text', value: text })
    }
    tokens.push({ type: 'expr', value: match[1].trim() })
    lastIndex = regex.lastIndex
  }
  // 最后的纯文本
  if (lastIndex < template.length) {
    tokens.push({ type: 'text', value: template.slice(lastIndex) })
  }

  // 将 Token 转换为代码
  tokens.forEach((token) => {
    if (token.type === 'text') {
      // 纯文本：直接拼接到 result（注意转义引号和换行）
      code += `result += ${JSON.stringify(token.value)};\n`
    } else {
      const expr = token.value
      if (expr.startsWith('#if ')) {
        code += `if (${expr.slice(4)}) {\n`
      } else if (expr.startsWith('#each ')) {
        const parts = expr.slice(6).split(' as ')
        const arr = parts[0].trim()
        const item = parts[1]?.trim() || 'item'
        code += `for (const ${item} of ${arr}) {\n`
      } else if (expr === '/if' || expr === '/each') {
        code += '}\n'
      } else {
        // 变量/表达式输出
        code += `result += ${expr};\n`
      }
    }
  })

  code += 'return result;'

  // 编译为渲染函数
  return new Function('data', `with(data){${code}}`)
}

// 使用示例
const tpl = `
  <div class="card">
    <h2>{{ title }}</h2>
    <ul>
      {{#each items as item }}
        <li class="{{ item.active ? 'active' : '' }}">
          {{ item.name }}
        </li>
      {{/each}}
    </ul>
    {{#if showFooter }}
      <footer>共 {{ items.length }} 项</footer>
    {{/if}}
  </div>
`

const render = compile(tpl)
const html = render({
  title: '商品列表',

    { name: '苹果', active: true },
    { name: '香蕉', active: false },
    { name: '橙子', active: true }
  ],
  showFooter: true
})

      threshold: 0.01
    }
  )

  images.forEach((img) => observer.observe(img))
}

    this.container.addEventListener('touchend', this.onTouchEnd.bind(this))
  }

  onTouchStart(e) {
    // 只在滚动到顶部时启用下拉
    if (this.container.scrollTop <= 0 && !this.refreshing) {
      this.startY = e.touches[0].pageY
    }
  }

  onTouchMove(e) {
    if (this.refreshing) return
    const moveY = e.touches[0].pageY - this.startY
    if (moveY > 0 && this.container.scrollTop <= 0) {
      this.pullDistance = Math.min(moveY * 0.5, 120) // 阻尼系数 0.5，最大 120px
      this.container.style.transform = `translateY(${this.pullDistance}px)`
      this.container.style.transition = 'none'
    }
  }

  onTouchEnd() {
    if (this.pullDistance >= this.threshold) {
      // 触发刷新
      this.refreshing = true
      this.container.style.transform = `translateY(50px)`
      this.container.style.transition = 'transform 0.3s'
      this.onRefresh().then(() => {
        this.refreshing = false
        this.container.style.transform = 'translateY(0)'
        this.container.style.transition = 'transform 0.3s'
      })
    } else {
      // 弹回
      this.container.style.transform = 'translateY(0)'
      this.container.style.transition = 'transform 0.3s'
    }
    this.pullDistance = 0
    this.startY = 0
  }
}

各数据结构特性与适用场景 ​

数组：连续内存，随机访问 O(1)。适合按索引频繁读取的场景。

  • 优点：下标访问极快、缓存友好（空间局部性）
  • 缺点：中间插入/删除需要移动元素 O(n)、扩容代价
  • 前端场景：DOM 列表、数据集合、TypedArray 处理二进制数据

链表：节点分散存储，通过指针连接。适合频繁增删首尾元素的场景。

  • 优点：插入/删除 O(1)（已知位置），不需要连续内存
  • 缺点：查找 O(n)，额外指针内存开销，缓存不友好
  • 前端场景：React Fiber 架构（链表实现的虚拟栈，可中断遍历）、LRU 缓存

栈：后进先出（LIFO），一端操作。适合"撤销/回溯"场景。

  • 前端场景：浏览器历史记录（前进后退）、函数调用栈、括号匹配、表达式求值、Undo/Redo

队列：先进先出（FIFO），两端操作。适合"排队/缓冲"场景。

  • 前端场景：事件循环的消息队列和微任务队列、动画队列、并发请求池、BFS 遍历 DOM

哈希表：键值对存储，通过哈希函数 O(1) 定位。适合"快速查找"场景。

  • 优点：查找/插入/删除均摊 O(1)
  • 缺点：无序、哈希碰撞退化、空间利用率低（负载因子）
  • 前端场景：Map 存缓存、Set 去重、WeakMap 存 DOM 关联数据、Object 作为字典

树：层级结构，根-父-子关系。适合"层级关系"场景。

  • 前端场景：DOM 树、组件树（Vue/React）、文件目录、路由配置树、AST 语法树

堆：完全二叉树，父节点总大于（最大堆）或小于（最小堆）子节点。适合"取极值"场景。

  • 前端场景：React Scheduler 优先级任务调度（最小堆维护 task 过期时间）、Top K 问题、合并 K 个排序列表

图：节点 + 边，不限关系。适合"网络关系"场景。

  • 前端场景：模块依赖图（Webpack 拓扑排序）、路由跳转图、社交关系图、地铁/地图路径规划

选择数据结构的决策思路 ​

需要快速按索引访问？ → 数组
需要快速查找键值？ → 哈希表（Map/Object）
需要层级关系？ → 树
需要后进先出/回溯？ → 栈
需要先进先出/排队？ → 队列
需要频繁取最大/最小值？ → 堆
需要表示网络/依赖关系？ → 图
需要在中间频繁增删？ → 链表

// 优化版：用对象 + 双指针实现 O(1) dequeue
class OptimizedQueue {
  constructor() {
    this.items = {}
    this.head = 0 // 队首指针
    this.tail = 0 // 队尾指针
  }
  enqueue(item) {
    this.items[this.tail++] = item
  }
  dequeue() {
    if (this.head === this.tail) return undefined
    const item = this.items[this.head]
    delete this.items[this.head++]
    // 定期清理：当指针偏移过大时重置
    if (this.head > 1000) {
      const arr = []
      for (let i = this.head; i < this.tail; i++) arr.push(this.items[i])
      this.items = {}
      for (let i = 0; i < arr.length; i++) this.items[i] = arr[i]
      this.tail = arr.length
      this.head = 0
    }
    return item
  }
  get size() {
    return this.tail - this.head
  }
}

哈希表（Hash Table） ​

结构：通过哈希函数将 key 映射到存储桶索引。冲突解决通常用拉链法（Chaining）或开放寻址法（Open Addressing）。

javascript
// 拉链法哈希表
class HashTable {
  constructor(size = 53) {
    this.table = new Array(size)
    this.size = size
  }

  _hash(key) {
    let total = 0
    const str = String(key)
    for (let i = 0; i < Math.min(str.length, 100); i++) {
      total = (total * 31 + str.charCodeAt(i)) % this.size
    }
    return total
  }

  set(key, value) {
    const index = this._hash(key)
    if (!this.table[index]) this.table[index] = []
    // 更新或追加
    const bucket = this.table[index]
    const existing = bucket.find(([k]) => k === key)
    if (existing) {
      existing[1] = value
    } else {
      bucket.push([key, value])
    }
  }

  get(key) {
    const index = this._hash(key)
    const bucket = this.table[index]
    if (!bucket) return undefined
    const entry = bucket.find(([k]) => k === key)
    return entry ? entry[1] : undefined
  }

  delete(key) {
    const index = this._hash(key)
    const bucket = this.table[index]
    if (!bucket) return false
    const idx = bucket.findIndex(([k]) => k === key)
    if (idx === -1) return false
    bucket.splice(idx, 1)
    return true
  }
}

    if (!this.root) {
      this.root = node
      return
    }
    let cur = this.root
    while (true) {
      if (val < cur.val) {
        if (!cur.left) {
          cur.left = node
          return
        }
        cur = cur.left
      } else {
        if (!cur.right) {
          cur.right = node
          return
        }
        cur = cur.right
      }
    }
  }

  search(val) {
    let cur = this.root
    while (cur) {
      if (val === cur.val) return cur
      cur = val < cur.val ? cur.left : cur.right
    }
    return null
  }

  // 三种遍历方式
  preOrder(node, cb) {
    // 根→左→右
    if (!node) return
    cb(node.val)
    this.preOrder(node.left, cb)
    this.preOrder(node.right, cb)
  }

  inOrder(node, cb) {
    // 左→根→右（BST 中序遍历得到有序序列）
    if (!node) return
    this.inOrder(node.left, cb)
    cb(node.val)
    this.inOrder(node.right, cb)
  }

  postOrder(node, cb) {
    // 左→右→根
    if (!node) return
    this.postOrder(node.left, cb)
    this.postOrder(node.right, cb)
    cb(node.val)
  }
}

class Graph {
  constructor() {
    this.adjacencyList = new Map() // 邻接表
  }

  addVertex(vertex) {
    if (!this.adjacencyList.has(vertex)) {
      this.adjacencyList.set(vertex, [])
    }
  }

  addEdge(v1, v2, directed = false) {
    this.adjacencyList.get(v1).push(v2)
    if (!directed) this.adjacencyList.get(v2).push(v1)
  }

  // BFS 最短路径（无权重图）
  bfs(start) {
    const visited = new Set()
    const queue = [start]
    const result = []
    visited.add(start)

    while (queue.length) {
      const vertex = queue.shift()
      result.push(vertex)
      for (const neighbor of this.adjacencyList.get(vertex)) {
        if (!visited.has(neighbor)) {
          visited.add(neighbor)
          queue.push(neighbor)
        }
      }
    }
    return result
  }

  // DFS
  dfs(start) {
    const visited = new Set()
    const result = []

    const traverse = (vertex) => {
      visited.add(vertex)
      result.push(vertex)
      for (const neighbor of this.adjacencyList.get(vertex)) {
        if (!visited.has(neighbor)) traverse(neighbor)
      }
    }

    traverse(start)
    return result
  }

  // 拓扑排序：返回依赖顺序，检测循环依赖
  topologicalSort() {
    const inDegree = new Map() // 入度
    const visited = new Set()

    // 初始入度
    for (const [v] of this.adjacencyList) {
      inDegree.set(v, 0)
      if (!visited.has(v)) visited.add(v)
    }
    for (const [, neighbors] of this.adjacencyList) {
      for (const n of neighbors) {
        inDegree.set(n, (inDegree.get(n) || 0) + 1)
      }
    }

    const queue = []
    const result = []

    // 入度为 0 的节点入队
    for (const [v, deg] of inDegree) {
      if (deg === 0) queue.push(v)
    }

    while (queue.length) {
      const v = queue.shift()
      result.push(v)
      for (const n of this.adjacencyList.get(v)) {
        inDegree.set(n, inDegree.get(n) - 1)
        if (inDegree.get(n) === 0) queue.push(n)
      }
    }

    // 结果数 ≠ 顶点数 → 有环
    if (result.length !== visited.size) {
      throw new Error('检测到循环依赖')
    }
    return result
  }
}

堆（Heap） ​

结构：完全二叉树（父子关系决定顺序），物理上用数组存储。

  • 最小堆：父节点 ≤ 子节点，根最小
  • 最大堆：父节点 ≥ 子节点，根最大

父子索引关系（数组下标从 0 开始）：

  • 父节点：Math.floor((i - 1) / 2)
  • 左子节点：2 * i + 1
  • 右子节点：2 * i + 2
javascript
class MinHeap {
  constructor() {
    this.heap = []
  }

  get size() {
    return this.heap.length
  }

  // 上浮（insert 时用）
  _bubbleUp(index) {
    while (index > 0) {
      const parent = Math.floor((index - 1) / 2)
      if (this.heap[parent] <= this.heap[index]) break
      ;[this.heap[parent], this.heap[index]] = [this.heap[index], this.heap[parent]]
      index = parent
    }
  }

  // 下沉（extractMin 时用）
  _bubbleDown(index) {
    const last = this.heap.length - 1
    while (true) {
      let smallest = index
      const left = 2 * index + 1
      const right = 2 * index + 2

      if (left <= last && this.heap[left] < this.heap[smallest]) smallest = left
      if (right <= last && this.heap[right] < this.heap[smallest]) smallest = right

      if (smallest === index) break
      ;[this.heap[index], this.heap[smallest]] = [this.heap[smallest], this.heap[index]]
      index = smallest
    }
  }

  insert(val) {
    this.heap.push(val)
    this._bubbleUp(this.heap.length - 1)
  }

  extractMin() {
    if (this.heap.length === 0) return null
    if (this.heap.length === 1) return this.heap.pop()

    const min = this.heap[0]
    this.heap[0] = this.heap.pop() // 末尾元素移到根
    this._bubbleDown(0)
    return min
  }

  peek() {
    return this.heap[0] ?? null
  }
}

// heapify: 将无序数组 O(n) 构建为堆
function heapify(arr) {
  const heap = new MinHeap()
  heap.heap = [...arr]
  // 从最后一个非叶子节点开始向下调整
  for (let i = Math.floor(arr.length / 2) - 1; i >= 0; i--) {
    heap._bubbleDown(i)
  }
  return heap
}

    j = 0

    result.push(left[i] <= right[j] ? left[i++] : right[j++])

}

快速排序（Quick Sort） ​

原理：分治——选一个 pivot，分区（partition）使左边都 ≤ pivot，右边都 ≥ pivot，再递归排序左右。

javascript
function quickSort(arr, left = 0, right = arr.length - 1) {
  if (left >= right) return arr

  const pivotIndex = partition(arr, left, right)
  quickSort(arr, left, pivotIndex - 1)
  quickSort(arr, pivotIndex + 1, right)
  return arr
}

function partition(arr, left, right) {
  // 取最后一个元素为 pivot
  const pivot = arr[right]
  let i = left // i 指向第一个大于 pivot 的位置

  for (let j = left; j < right; j++) {
    if (arr[j] < pivot) {
      ;[arr[i], arr[j]] = [arr[j], arr[i]]
      i++
    }
  }
  ;[arr[i], arr[right]] = [arr[right], arr[i]] // pivot 归位
  return i
}

function heapSort(arr) {
  const n = arr.length

  // 建最大堆（原地，从最后一个非叶子节点开始）
  for (let i = Math.floor(n / 2) - 1; i >= 0; i--) {
    heapify(arr, n, i)
  }

  // 逐一将最大值移到末尾
  for (let i = n - 1; i > 0; i--) {
    ;[arr[0], arr[i]] = [arr[i], arr[0]] // 堆顶（最大值）放到末尾
    heapify(arr, i, 0) // 剩余部分重新堆化
  }
  return arr
}

function heapify(arr, size, root) {
  let largest = root
  const left = 2 * root + 1
  const right = 2 * root + 2

  if (left < size && arr[left] > arr[largest]) largest = left
  if (right < size && arr[right] > arr[largest]) largest = right

  if (largest !== root) {
    ;[arr[root], arr[largest]] = [arr[largest], arr[root]]
    heapify(arr, size, largest)
  }
}

4.3 了解递归和循环的优缺点、应用场景、并可在开发中熟练应用 ​

参考

递归和循环是两种最基本的控制结构。递归适合自相似、可分解的问题；循环适合线性重复的问题。两者可以互相转化。

递归的本质 ​

递归由两部分组成：

  1. 基准条件（Base Case）：终止递归的条件
  2. 递归步骤（Recursive Step）：将问题分解为规模更小的子问题

每次递归调用都会向调用栈压入一个新帧，因此有栈溢出风险。

javascript

function factorial(n) {
  if (n <= 1) return 1 // base case
  return n * factorial(n - 1) // recursive step
}

// 等价循环实现
function factorialLoop(n) {
  let result = 1
  for (let i = 2; i <= n; i++) result *= i
  return result
}

尾递归优化（Tail Call Optimization, TCO） ​

当一个函数的最后一步操作是调用自身时，引擎可以直接复用当前栈帧，不增加栈深度。

// 尾递归改写（最后一步就是调用自身）
function factorialTail(n, acc = 1) {
  if (n <= 1) return acc
  return factorialTail(n - 1, acc * n) // 纯尾调用，无待处理运算

// 将递归改为返回"执行下一步"的函数，用循环驱动
function trampoline(fn) {
  return function (...args) {
    let result = fn(...args)
    while (typeof result === 'function') {
      result = result() // 循环中调用，不增长栈
    }
    return result
  }
}

function factorialTramp(n, acc = 1) {
  if (n <= 1) return acc
  // 返回一个 thunk，而不是直接递归调用
  return () => factorialTramp(n - 1, acc * n)
}

回溯算法（Backtracking） ​

核心思想：系统地搜索所有可能的解，在搜索过程中一旦发现当前选择不可能得到解就"回溯"（撤销选择），尝试其他路径。本质是深度优先搜索 + 剪枝。

模板：

javascript
function backtrack(路径, 选择列表) {
  if (满足结束条件) {
    result.push([...路径]);
    return;
  }
  for (const 选择 of 选择列表) {
    做选择 → 路径.push(选择);
    backtrack(路径, 新选择列表); // 递归
    撤销选择 → 路径.pop();
  }
}

// 全排列问题
function permute(nums) {
  const result = []
  const used = new Array(nums.length).fill(false)

  function backtrack(path) {
    if (path.length === nums.length) {
      result.push([...path])
      return
    }
    for (let i = 0; i < nums.length; i++) {
      if (used[i]) continue
      path.push(nums[i])
      used[i] = true
      backtrack(path)
      path.pop() // 回溯：撤销选择
      used[i] = false
    }
  }

  backtrack([])
  return result
}

function coinChange(coins, amount) {

  const result = []
  for (const coin of coins) {
    while (amount >= coin) {
      amount -= coin
      result.push(coin)
    }
  }
  return amount === 0 ? result : null // 无解返回 null
}

// 二分查找：分治思维
function binarySearch(arr, target, left = 0, right = arr.length - 1) {
  if (left > right) return -1
  const mid = Math.floor((left + right) / 2)
  if (arr[mid] === target) return mid
  if (arr[mid] > target) return binarySearch(arr, target, left, mid - 1)
  return binarySearch(arr, target, mid + 1, right)
}

// 快速幂（pow）：分治 O(log n)
function pow(x, n) {
  if (n === 0) return 1
  if (n < 0) return 1 / pow(x, -n)
  const half = pow(x, Math.floor(n / 2))
  return n % 2 === 0 ? half * half : half * half * x
}

// 斐波那契：从递归到 DP 的演进

// 1. 纯递归 O(2ⁿ) — 大量重复计算
function fib(n) {
  if (n <= 1) return n
  return fib(n - 1) + fib(n - 2)
}

// 2. 记忆化（自顶向下）O(n) — 缓存中间结果
function fibMemo(n, memo = {}) {
  if (n <= 1) return n
  if (memo[n] !== undefined) return memo[n]
  memo[n] = fibMemo(n - 1, memo) + fibMemo(n - 2, memo)
  return memo[n]
}

// 3. 递推/DP表（自底向上）O(n) O(1)
function fibDP(n) {
  if (n <= 1) return n
  let prev2 = 0,
    prev1 = 1
  for (let i = 2; i <= n; i++) {
    const cur = prev1 + prev2
    prev2 = prev1
    prev1 = cur
  }
  return prev1
}

// 01背包问题
function knapsack(weights, values, capacity) {
  const n = weights.length
  // dp[i][c] = 前 i 件物品，容量 c 下的最大价值
  const dp = Array.from({ length: n + 1 }, () => new Array(capacity + 1).fill(0))

  for (let i = 1; i <= n; i++) {
    for (let c = 0; c <= capacity; c++) {
      if (weights[i - 1] > c) {
        dp[i][c] = dp[i - 1][c] // 装不下
      } else {
        dp[i][c] = Math.max(
          dp[i - 1][c], // 不装
          dp[i - 1][c - weights[i - 1]] + values[i - 1] // 装
        )
      }
    }
  }
  return dp[n][capacity]
}

方案一：时间分片（Time Slicing） ​

将长任务拆分为多个小任务，穿插在事件循环的空闲中执行，保证 UI 不卡顿。

javascript
// 使用 requestIdleCallback（适合非紧急任务）
function processInIdle(items, processFn, doneFn) {
  let i = 0
  function processChunk(deadline) {
    // 当有空闲时间且还有未处理的数据时持续执行
    while (i < items.length && deadline.timeRemaining() > 1) {
      processFn(items[i], i)
      i++
    }
    if (i < items.length) {
      requestIdleCallback(processChunk)
    } else if (doneFn) {
      doneFn()
    }
  }
  requestIdleCallback(processChunk)
}

// 使用 setTimeout 分片（兼容性最好）
function timeSlice(items, processFn, chunkSize = 50, doneFn) {
  let i = 0
  function chunk() {
    const end = Math.min(i + chunkSize, items.length)
    for (; i < end; i++) {
      processFn(items[i], i)
    }
    if (i < items.length) {
      setTimeout(chunk, 0) // 将控制权还给主线程
    } else if (doneFn) {
      doneFn()
    }
  }
  setTimeout(chunk, 0)
}

// 使用 scheduler.yield()（现代浏览器，取代 setTimeout）
async function processAsync(items, processFn, chunkSize = 50) {
  for (let i = 0; i < items.length; i++) {
    processFn(items[i], i)
    if (i % chunkSize === 0 && 'scheduler' in window) {
      await window.scheduler.yield() // 让出主线程
    }
  }
}

class VirtualList {
  constructor(container, itemHeight, totalItems, renderItem) {
    this.container = container
    this.itemHeight = itemHeight
    this.totalItems = totalItems
    this.renderItem = renderItem

    // 创建占位元素撑开滚动条
    this.placeholder = document.createElement('div')
    this.placeholder.style.height = totalItems * itemHeight + 'px'
    this.placeholder.style.position = 'relative'

    // 创建可视区域容器
    this.viewport = document.createElement('div')
    this.viewport.style.position = 'absolute'
    this.viewport.style.top = '0'
    this.viewport.style.left = '0'
    this.viewport.style.right = '0'

    this.container.appendChild(this.placeholder)
    this.placeholder.appendChild(this.viewport)

    this.render()
    this.container.addEventListener('scroll', () => this.render())
  }

  render() {
    const scrollTop = this.container.scrollTop
    const containerHeight = this.container.clientHeight
    const buffer = 5 // 上下各多渲染5项，防止白屏

    const startIndex = Math.max(0, Math.floor(scrollTop / this.itemHeight) - buffer)
    const endIndex = Math.min(
      this.totalItems,
      Math.ceil((scrollTop + containerHeight) / this.itemHeight) + buffer
    )

    this.viewport.style.transform = `translateY(${startIndex * this.itemHeight}px)`

    // 简单实现：直接重建 DOM（生产环境建议用 DOM 复用或 React/Vue 的 key 缓存）
    this.viewport.innerHTML = ''
    for (let i = startIndex; i < endIndex; i++) {
      this.viewport.appendChild(this.renderItem(i))
    }
  }
}

  }

时间分片列表/表格数据批量渲染1万-10万requestIdleCallback / scheduler.yield任务必须可拆分，单次执行 < 16ms
虚拟列表长列表/表格/聊天记录1万-100万scroll + transform:translateY动态高度场景需维护位置缓存 + 二分查找
Web Worker排序/搜索/转换/解析不限Worker + Transferable不能操作DOM；数据传输有序列化开销
Canvas/WebGL图表/散点/热力图10万+Canvas 2D / Three.js / ECharts GL不支持无障碍；文字渲染精度不如 DOM
服务端分页数据库级别的海量数据百万级接口分页 + 前端缓存依赖网络延迟；搜索/筛选需要在后端实现

实际综合案例：100万条记录的管理系统 ​

javascript
// 组合方案：Worker排序 + 虚拟列表渲染

// 1. Worker 接收全量数据，排序、筛选全在 Worker 做
const worker = new Worker('data-worker.js')

// 2. 主线程用虚拟列表渲染 Worker 返回的数据
class DataTable {
  constructor(data) {
    this.worker = new Worker('data-worker.js')
    this.virtualList = new VirtualList(/* ... */)

    this.worker.onmessage = (e) => {
      // Worker 返回处理后的数据子集
      this.updateViewport(e.data)
    }
  }

  // 排序或筛选时，告诉 Worker 重新计算
  sort(column, order) {
    this.worker.postMessage({ action: 'sort', column, order })
  }

  filter(predicate) {
    this.worker.postMessage({ action: 'filter', predicate })
  }

  updateViewport(visibleData) {
    // 只更新可见区域
    this.virtualList.update(visibleData)
  }
}

总结：海量数据处理的本质是不让主线程处理超过约 50ms 的连续任务（保持 60fps）。策略就是减少工作量（虚拟列表）、拆散工作（时间分片）、转移工作（Worker）、减少工作产出（数据聚合）。 :::
