# js随机彩票

## 代码示例

```
const rendom = (max, min = 1) => ~~(Math.random()*(max-min))+min

const getArr = (max, len, arr = []) => {
  while(arr.length<len){
    const num = rendom(max)
    !arr.includes(num) && arr.push(num)
  }
  return arr.sort((a, b) => a - b)
}

console.log([...getArr(35, 5), ...getArr(12, 2)]);
console.log([...getArr(33, 6), rendom(16)]);
```
